cmake_minimum_required(VERSION 3.18)
project(classlive_offline LANGUAGES C CXX)
if(NOT EXISTS "${TRANSCRIBE_SOURCE}/include/transcribe.h")
    message(FATAL_ERROR "Set TRANSCRIBE_SOURCE to the pinned Hex transcribe.cpp checkout")
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(TRANSCRIBE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_USE_SYSTEM_BLAS OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_VULKAN OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CCACHE OFF CACHE BOOL "" FORCE)
add_subdirectory("${TRANSCRIBE_SOURCE}" transcribe-build EXCLUDE_FROM_ALL)
target_include_directories(transcribe PUBLIC "${TRANSCRIBE_SOURCE}/include")
if(NOT ANDROID AND NOT JNI_INCLUDE_DIRS)
    find_package(JNI REQUIRED)
endif()
add_library(classlive_asr SHARED bridge.cpp)
target_include_directories(classlive_asr PRIVATE ${JNI_INCLUDE_DIRS})
target_link_libraries(classlive_asr PRIVATE transcribe)
